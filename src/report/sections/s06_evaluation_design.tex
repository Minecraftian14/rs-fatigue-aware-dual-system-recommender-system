\section{Evaluation Design}

Evaluation is done in a simulation loop over sampled users and time steps. 
At each step, the model produces recommendations, the simulated user reacts to the top item, and the metrics are accumulated. 
This way allows observing how a recommender performs over time, where preference keeps changing and the user can get fatigued or discontented.

Let $R_u^k$ be the top-$k$ recommended items for user $u$, and let $T_u$ be the set of true relevant items for that user.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\subsection{Core Recommendation Accuracy}


\noindent \textbf{Hit Rate @ K} \\
Checks whether at least one relevant item appears in the top-$k$ list.

\begin{equation}
    \mathrm{HR@K}=\frac{1}{|U|}\sum_{u\in U}\mathbf{1}\left(|R_u^k\cap T_u|>0\right)
\end{equation}


\noindent \textbf{Precision @ K} \\
Measures the fraction of the returned top-$k$ items that are relevant, using the deduplicated top-$k$ set.

\begin{equation}
    \mathrm{Precision@K}=\frac{1}{|U|}\sum_{u\in U}\frac{|R_u^k\cap T_u|}{|R_u^k|}
\end{equation}


\noindent \textbf{Recall @ K} \\
Measures how much of the relevant set is recovered in the top-$k$ list.

\begin{equation}
    \mathrm{Recall@K}=\frac{1}{|U|}\sum_{u\in U}\frac{|R_u^k\cap T_u|}{\min(k,;|T_u|)}
\end{equation}


\noindent \textbf{NDCG @ K} \\
Gives higher credit to relevant items appearing earlier in the ranking.

\begin{equation}
    \mathrm{DCG@K}=\sum_{i=1}^{|R_u^k|}\frac{\mathbf{1}(r_i\in T_u)}{\log_2(i+1)}
\end{equation}

\begin{equation}
    \mathrm{NDCG@K}=\frac{\mathrm{DCG@K}}{\mathrm{IDCG@K}}
\end{equation}

where $\mathrm{IDCG@K}$ is the best possible DCG for that user.


\noindent \textbf{MRR @ K} \\
Measures how early the first relevant item appears in the ranked list.

\begin{equation}
    \mathrm{MRR@K}=\frac{1}{|U|}\sum_{u\in U}\frac{1}{\mathrm{rank}_u}
\end{equation}

where $\mathrm{rank}_u$ is the position of the first relevant item in the top-$k$ list, and $0$ if none is present.

\subsection{Long-Term Engagement and User Welfare}


\noindent \textbf{Session Depth} \\
Counts how many steps the user stays active in a session without becoming fatigued.

\begin{equation}
    \mathrm{SessionDepth}_u=\sum_{t=1}^{T}\mathbf{1}(\text{user not fatigued at } t)
\end{equation}


\noindent \textbf{Return Rate} \\
Measures how often a fatigued user comes back after a temporary break.

\begin{equation}
    \mathrm{ReturnRate}_u=\sum_{t=1}^{T}\mathbf{1}(\text{user returns after fatigue})
\end{equation}


\noindent \textbf{Time-to-Boredom} \\
Measures the average number of steps before fatigue interrupts the session.

\begin{equation}
    \mathrm{TimeToBoredom}_u=\frac{1}{m}\sum_{j=1}^{m} \ell_j
\end{equation}

where $\ell_j$ is the length of the $j$-th active stretch. 


\noindent \textbf{Time-to-Discontent} \\
Measures the average number of steps before discontentment stops interaction.

\begin{equation}
    \mathrm{TimeToDiscontent}_u=\frac{1}{n}\sum_{j=1}^{n} d_j
\end{equation}

where $d_j$ is the length of the $j$-th uninterrupted period before discontentment appears. 


\noindent \textbf{Acceptance Rate} \\
Measures how often the shown recommendations are clicked.

\begin{equation}
    \mathrm{AcceptanceRate}_u=\frac{1}{T}\sum_{t=1}^{T}\mathbf{1}(\text{click}_t)
\end{equation}

\subsection{Fatigue and Boredom Metrics}


\noindent \textbf{Content Repetition Rate} \\
Measures how similar consecutive recommended item embeddings are on average.

\begin{equation}
    \mathrm{RepetitionRate}_u=\frac{1}{T-1}\sum_{t=1}^{T-1}\cos(e_t,e_{t+1})
\end{equation}


\noindent \textbf{Novelty} \\
Measures how different recent recommendations are from each other.

\begin{equation}
    \mathrm{Novelty}_u=\frac{1}{T-1}\sum_{t=1}^{T-1}\left(1-\cos(e_t,e_T)\right)
\end{equation}


\noindent \textbf{Intra-List Diversity} \\
Measures how diverse the recent recommendation window is by looking at pairwise similarity inside that window.

\begin{equation}
    \mathrm{ILD}_u=1-\frac{1}{m(m-1)}\sum_{i\neq j}\cos(e_i,e_j)
\end{equation}


\noindent \textbf{Fatigue Score} \\
A simple composite signal built from boredom, novelty, and diversity.

\begin{equation}
    \mathrm{FatigueScore}_u=\mathrm{TimeToBoredom}_u-\mathrm{Novelty}_u-\mathrm{ILD}_u
\end{equation}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\subsection{Human-Centered Recommendation Metrics}


\noindent \textbf{Diversity} \\
Measures how diverse the recommendations are by looking at every two consequtive recommendations.

\begin{equation}
    \mathrm{Diversity}_u=\mathrm{ILD}_u
\end{equation}


\noindent \textbf{Serendipity} \\
Combines novelty with repetition-like behavior in the implemented code.

\begin{equation}
    \mathrm{Serendipity}_u=\mathrm{Novelty}_u+\mathrm{RepetitionRate}_u
\end{equation}


\noindent \textbf{Coverage} \\
Measures how much of the true-item set is touched by the recommendations over the evaluation run.

\begin{equation}
    \mathrm{Coverage}_u=\frac{|\mathcal{R}_u|}{|\mathcal{T}_u|}
\end{equation}

where $\mathcal{R}_u$ is the set of unique recommended items and $\mathcal{T}_u$ is the set of unique true items collected during evaluation.

\subsection{Baselines and Ablation Study}

To properly evaluate the proposed system, we compare it against several baseline models. 
These baselines are chosen to represent different levels of complexity, from classical recommendation methods to components of our own system.

We include the following models:

\begin{itemize}
    \item \textbf{Alternating Least Squares (ALS)} \\
        A classical collaborative filtering method based on matrix factorization. 
        It learns user and item latent factors by minimizing reconstruction error on observed interactions. 
        This serves as a strong traditional baseline.
    \item \textbf{Bayesian Personalized Ranking (BPR)} \\
        A pairwise ranking-based approach that learns to rank observed interactions higher than unobserved ones. 
        It is widely used for implicit feedback settings and provides a good comparison for ranking quality.
    \item \textbf{T0 Embeddings of System 1 (S1)} \\
        This is our two-tower model trained on the T0 dataset. 
        It represents a strong neural baseline that captures user preference without any temporal dynamics. 
    \item \textbf{System 2 Preference Model (S2P)} \\
        This model uses interaction history to update user embeddings. 
        It captures preference drift but does not consider fatigue or discontentment. 
        This baseline isolates the benefit of temporal modeling alone.
    \item \textbf{System 2 Cognitive Model (S2F)} \\
        This is the full reinforcement learning-based model. 
        It incorporates both preference evolution and cognitive states such as fatigue and discontentment. 
        This represents the most complete version of our system.
\end{itemize}
